apiVersion: apps/v1
kind: Deployment
metadata:
  name: spark-thrift
spec:
  replicas: 1
  selector:
    matchLabels:
      app: spark-thrift
  template:
    metadata:
      labels:
        app: spark-thrift
    spec:
      containers:
      - name: spark-thrift
        image: spark:3.4.1
        command:
        - /bin/bash
        - -c
        - |
          /opt/spark/sbin/start-thriftserver.sh \
            --master k8s://https://kubernetes.default.svc \
            --conf spark.sql.catalogImplementation=hive \
            --conf spark.hadoop.hive.metastore.uris=thrift://hive-metastore:9083 \
            --conf spark.sql.extensions=io.delta.sql.DeltaSparkSessionExtension \
            --conf spark.sql.catalog.spark_catalog=org.apache.spark.sql.delta.catalog.DeltaCatalog \
            --conf spark.sql.warehouse.dir=hdfs:///warehouse
        ports:
        - containerPort: 10000





apiVersion: v1
kind: Service
metadata:
  name: spark-thrift
spec:
  selector:
    app: spark-thrift
  ports:
  - port: 10000
    targetPort: 10000



apiVersion: batch/v1
kind: Job
metadata:
  name: register-delta-tables
spec:
  template:
    spec:
      containers:
      - name: spark-sql
        image: spark:3.4.1
        command:
        - /bin/bash
        - -c
        - |
          /opt/spark/bin/spark-sql <<EOF
          CREATE DATABASE IF NOT EXISTS st_db;

          CREATE TABLE IF NOT EXISTS st_db.orders
          USING DELTA
          LOCATION 'hdfs:///delta/st/orders';

          CREATE TABLE IF NOT EXISTS st_db.customers
          USING DELTA
          LOCATION 'hdfs:///delta/st/customers';
          EOF
      restartPolicy: Never






apiVersion: batch/v1
kind: Job
metadata:
  name: beeline-query
spec:
  template:
    spec:
      containers:
      - name: beeline
        image: spark:3.4.1
        command:
        - /bin/bash
        - -c
        - |
          /opt/spark/bin/beeline \
            -u jdbc:hive2://spark-thrift:10000 \
            -e "SELECT count(*) FROM st_db.orders;"
      restartPolicy: Never
